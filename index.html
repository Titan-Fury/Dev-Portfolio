async def compose(bundle: ContextBundle):
    prompt = select_prompt(bundle.trigger.kind)
    try:
        draft = await asyncio.wait_for(
            llm.generate(prompt, bundle),
            timeout=8.0,
        )
    except Exception:
        return fallback_compose(bundle)

    # defense-in-depth: strip / cap / dedupe
    draft = strip_urls(draft)
    draft = enforce_length(draft, 320)
    if is_repeat(draft, history):
        return Composition(action="end")
    return Composition(text=draft)
